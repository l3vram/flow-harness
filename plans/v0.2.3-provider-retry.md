## `src/retry.ts`
```
import type { FetchLike } from "./providers/openai-compatible.js";

export interface RetryOptions { maxRetries?: number; baseMs?: number; }

const RETRYABLE = new Set([408, 429, 500, 502, 503, 504]);

/**
 * Calls `fetchImpl`, retrying on transient HTTP statuses (429/408/5xx) with exponential backoff
 * + jitter, up to `maxRetries` times. Returns the final response; the caller still checks `.ok`.
 * A non-retryable status (e.g. 400/401/404) returns immediately, and a 2xx returns immediately.
 */
export async function fetchWithRetry(
  fetchImpl: FetchLike,
  url: string,
  init: { method: string; headers: Record<string, string>; body: string },
  options: RetryOptions = {},
): Promise<Awaited<ReturnType<FetchLike>>> {
  const maxRetries = options.maxRetries ?? 3;
  const baseMs = options.baseMs ?? 500;
  let res = await fetchImpl(url, init);
  let attempt = 0;
  while (!res.ok && RETRYABLE.has(res.status) && attempt < maxRetries) {
    const delay = baseMs * 2 ** attempt + Math.floor(Math.random() * baseMs);
    if (delay > 0) await new Promise((r) => setTimeout(r, delay));
    attempt++;
    res = await fetchImpl(url, init);
  }
  return res;
}
```
(Note: `FetchLike` is exported from `providers/openai-compatible.ts`.)

## `src/providers/openai-compatible.ts` and `src/providers/anthropic.ts`
- Add `maxRetries?: number` and `retryBaseMs?: number` to each provider's options interface; store
  them (defaults: `maxRetries` 3, `retryBaseMs` 500).
- Replace the direct `await this.fetchImpl(url, init)` call with
  `await fetchWithRetry(this.fetchImpl, url, init, { maxRetries: this.maxRetries, baseMs: this.retryBaseMs })`.
  Everything else (headers, body, response parsing, the `!res.ok` throw) is unchanged.

## Tests
### `test/retry.test.ts`
Use a fake `FetchLike` that returns a scripted queue of `{ ok, status }` responses and counts calls.
- 503 then 200 (with `baseMs: 0`) → returns the 200, made 2 calls.
- always 503, `maxRetries: 2`, `baseMs: 0` → returns a 503 after 3 calls (initial + 2 retries).
- 400 → returns immediately after 1 call (not retryable).
- 200 → 1 call.
